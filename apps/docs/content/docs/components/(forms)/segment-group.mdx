## Usage
A compound component used for selecting a single option from a set of options.

```tsx
import { SegmentGroup } from "@moto-ui/react";

<SegmentGroup.Root defaultValue="daily">
  <SegmentGroup.Indicator />
  <SegmentGroup.Item value="daily">
    <SegmentGroup.ItemText>Daily</SegmentGroup.ItemText>
    <SegmentGroup.ItemControl />
    <SegmentGroup.ItemHiddenInput />
  </SegmentGroup.Item>
  <SegmentGroup.Item value="weekly">
    <SegmentGroup.ItemText>Weekly</SegmentGroup.ItemText>
    <SegmentGroup.ItemControl />
    <SegmentGroup.ItemHiddenInput />
  </SegmentGroup.Item>
</SegmentGroup.Root>;
```
