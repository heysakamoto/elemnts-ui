## Usage
Used to select a single value from a list of options using a wheel-like interaction.


```tsx
import { WheelPicker } from "@elemnts-ui/react"

const options = [
  { label: "Gaming", value: "gaming" },
  { label: "Karate", value: "karate" },
  { label: "Cooking", value: "cooking" },
]

<WheelPicker>
  <WheelPicker.Control>
    <WheelPicker.Options options={options} defaultValue="gaming" />
  </WheelPicker.Control>
</WheelPicker>
```
