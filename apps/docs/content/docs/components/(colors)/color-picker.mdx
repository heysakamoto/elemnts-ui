## Usage

A feature-rich compound component that provides a complete interface for color selection, including color areas, sliders, and swatches.

```tsx
import {
  ColorPicker,
  Swatch,
  Surface,
  Portal,
  parseColor,
} from "@elemnts-ui/react";

const color = parseColor("#eb5e41");

<ColorPicker defaultValue={color}>
  <ColorPicker.Control>
    <ColorPicker.Trigger>
      <ColorPicker.ValueSwatch asChild>
        <Swatch size="sm" />
      </ColorPicker.ValueSwatch>
    </ColorPicker.Trigger>
  </ColorPicker.Control>
  <Portal>
    <ColorPicker.Positioner>
      <ColorPicker.Content>
        <Surface p="8">
          <ColorPicker.Area>
            <ColorPicker.AreaBackground>
              <ColorPicker.AreaThumb />
            </ColorPicker.AreaBackground>
          </ColorPicker.Area>
          <ColorPicker.ChannelSlider channel="hue">
            <ColorPicker.ChannelSliderTrack>
              <ColorPicker.ChannelSliderThumb />
            </ColorPicker.ChannelSliderTrack>
          </ColorPicker.ChannelSlider>
        </Surface>
      </ColorPicker.Content>
    </ColorPicker.Positioner>
  </Portal>
</ColorPicker>;
```
